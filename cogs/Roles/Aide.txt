# ============================================================
# 🐍 EXCEPTIONS PYTHON
# ============================================================

# ValueError
# → La valeur est incorrecte, même si le type est correct.
# Exemple : int("bonjour")
except ValueError:
    pass


# TypeError
# → Le type de donnée n'est pas adapté.
# Exemple : "5" + 10
except TypeError:
    pass


# KeyError
# → Une clé demandée n'existe pas dans un dictionnaire.
# Exemple : data["nom"] alors que "nom" n'existe pas.
except KeyError:
    pass


# IndexError
# → On demande un élément qui n'existe pas dans une liste.
# Exemple : liste[10] alors que la liste contient 3 éléments.
except IndexError:
    pass


# AttributeError
# → L'objet ne possède pas l'attribut demandé.
# Exemple : membre.truc_inexistant
except AttributeError:
    pass


# FileNotFoundError
# → Le fichier demandé n'existe pas.
except FileNotFoundError:
    pass


# PermissionError
# → Python n'a pas la permission d'accéder au fichier/dossier.
except PermissionError:
    pass


# OSError
# → Erreur générale liée au système/fichiers.
except OSError:
    pass


# ============================================================
# 🌐 EXCEPTIONS DISCORD.PY
# ============================================================

import discord


# discord.Forbidden
# → Discord refuse l'action car le bot n'a pas les permissions.
# Exemple : ajouter/modifier/supprimer un rôle sans permission.
except discord.Forbidden:
    pass


# discord.NotFound
# → La ressource Discord demandée n'existe pas.
# Exemple : membre, rôle, salon ou message introuvable.
except discord.NotFound:
    pass


# discord.HTTPException
# → Discord a renvoyé une erreur HTTP.
# Exemple : problème avec l'API Discord.
except discord.HTTPException:
    pass


# discord.LoginFailure
# → Le bot n'arrive pas à se connecter.
# Exemple : token invalide.
except discord.LoginFailure:
    pass


# ============================================================
# 🗄️ SQLITE
# ============================================================

import sqlite3


# sqlite3.Error
# → Erreur générale SQLite.
except sqlite3.Error:
    pass


# sqlite3.OperationalError
# → Problème d'exécution SQL / base de données.
# Exemple : table inexistante, DB verrouillée, requête incorrecte.
except sqlite3.OperationalError:
    pass


# sqlite3.IntegrityError
# → Violation d'une contrainte SQL.
# Exemple : PRIMARY KEY / UNIQUE.
except sqlite3.IntegrityError:
    pass
